# Phase 1 — Load Profile

Load the user's health history, build a timeline, and identify research topics.

```ts
const __flow = tasklist("review", {
  load_profile: {
    description: "Load user's health history",
    outputSchema: { type: "object", required: ["history", "profileExists"], properties: { history: { type: "object" }, profileExists: { type: "boolean" } } },
  },
  build_timeline: {
    description: "Build chronological timeline",
    dependsOn: ["load_profile"],
    outputSchema: { type: "object", required: ["events"], properties: { events: { type: "array" } } },
  },
  identify_topics: {
    description: "Identify topics for research",
    dependsOn: ["load_profile"],
    outputSchema: { type: "object", required: ["topics"], properties: { topics: { type: "array", items: { type: "string" } } } },
  },
  research_topics: {
    description: "Research identified topics",
    dependsOn: ["identify_topics"],
    optional: true,
    outputSchema: { type: "object", properties: { researchResults: { type: "array" } } },
  },
  generate_insights: {
    description: "Generate personalized insights",
    dependsOn: ["build_timeline", "research_topics"],
    outputSchema: { type: "object", required: ["insights"], properties: { insights: { type: "array" } } },
  },
  compose_review: {
    description: "Compose health review",
    dependsOn: ["generate_insights"],
    outputSchema: { type: "object", required: ["review"], properties: { review: { type: "object" } } },
  },
  submit: {
    description: "Submit the review",
    dependsOn: ["compose_review"],
  },
});

// ── load_profile ──────────────────────────────────────────────────
__flow.start("load_profile");
const history = await loadHistory() as HealthHistory;
pin("history");
const profileExists = history.totalRecords > 0;
__flow.finish("load_profile", { history, profileExists });

// ── build_timeline ────────────────────────────────────────────────
__flow.start("build_timeline");
const events = Object.entries(history.categories).flatMap(([cat, records]) =>
  records.map(r => ({ date: r.date as string, type: cat, title: r.title ?? cat, details: r }))
).sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime());

display(<HealthTimeline events={events} />);
__flow.finish("build_timeline", { events });

// ── identify_topics ───────────────────────────────────────────────
__flow.start("identify_topics");
// Analyze profile for topics needing research:
// - Abnormal lab values
// - New medications
// - Chronic conditions
// - Recent changes
const topics: string[] = [];
// Inspect history.categories for concerning patterns
__flow.finish("identify_topics", { topics });

// ── yield ─────────────────────────────────────────────────────────
await inspect({ history, events, topics });
```
