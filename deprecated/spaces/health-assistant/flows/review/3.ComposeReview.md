# Phase 3 — Generate Insights, Compose Review, Submit

Synthesize all gathered data into a comprehensive health review.

```ts
// ── generate_insights ─────────────────────────────────────────────
__flow.start("generate_insights");
const insights: Array<{ category: string; finding: string; severity: string; recommendation: string }> = [];
// Analyze history + researchResults to generate insights
// Flag abnormal lab values, concerning trends, medication interactions, etc.
__flow.finish("generate_insights", { insights });

// ── compose_review ────────────────────────────────────────────────
__flow.start("compose_review");
const review = {
  generatedAt: new Date().toISOString(),
  profileSummary: `Total records: ${history.totalRecords}, Categories: ${Object.keys(history.categories).join(", ")}`,
  insights,
  recommendations: insights.map(i => i.recommendation),
};
__flow.finish("compose_review", { review });

// ── submit ────────────────────────────────────────────────────────
__flow.start("submit");
submitReview({
  generatedAt: review.generatedAt,
  profile: history,
  insights,
  timeline: events,
  recommendations: review.recommendations,
  disclaimer: "This health review is for informational purposes only. It does not constitute medical advice. Consult a healthcare professional for medical decisions.",
});
```
