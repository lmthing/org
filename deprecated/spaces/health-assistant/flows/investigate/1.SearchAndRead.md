# Phase 1 — Plan, Search, and Read

Decompose the question, delegate search and reading to the research space.

```ts
const __flow = tasklist("investigate", {
  plan: {
    description: "Decompose question and select sources",
    outputSchema: { type: "object", required: ["queries", "sources"], properties: { queries: { type: "array", items: { type: "string" } }, sources: { type: "array", items: { type: "string" } } } },
  },
  delegate_search: {
    description: "Delegate search to research space",
    dependsOn: ["plan"],
    outputSchema: { type: "object", required: ["searchResults"], properties: { searchResults: { type: "array" } } },
  },
  delegate_read: {
    description: "Delegate reading to research space",
    dependsOn: ["delegate_search"],
    outputSchema: { type: "object", required: ["excerpts"], properties: { excerpts: { type: "array" } } },
  },
  submit: {
    description: "Synthesize and submit",
    dependsOn: ["delegate_read"],
  },
});

// ── plan ──────────────────────────────────────────────────────────
__flow.start("plan");
// Decompose the user's question into targeted queries
const queries = [userQuestion]; // Expand into 1-3 sub-queries
const sources = ["pubmed", "nih", "mayo_clinic"];
__flow.finish("plan", { queries, sources });

// ── delegate_search ───────────────────────────────────────────────
__flow.start("delegate_search");
const searchResults: Array<{ output: string; status: string }> = [];
for (const query of queries) {
  const result = await delegate({
    space: "research",
    agent: "searcher",
    flow: "search",
    task: `Search for: "${query}". Focus on trusted medical sources. Return top 8 results.`,
  }) as { output: string; status: "ok" | "error" };
  searchResults.push(result);
}
__flow.finish("delegate_search", { searchResults });

// ── delegate_read ─────────────────────────────────────────────────
__flow.start("delegate_read");
const excerpts: string[] = [];
// Extract top URLs from search results and delegate reading
// For each top URL:
//   const readResult = await delegate({ space: "research", agent: "reader", flow: "read", task: `Read: ${url}. Byte budget: 30000.` });
//   excerpts.push(readResult.output);
__flow.finish("delegate_read", { excerpts });

// ── yield ─────────────────────────────────────────────────────────
await inspect({ queries, sources, searchResults, excerpts });
```
