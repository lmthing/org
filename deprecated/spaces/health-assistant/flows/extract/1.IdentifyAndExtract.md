# Phase 1 — Identify and Extract

Classify the document type, load extraction guidance, and extract structured data.

```ts
const __flow = tasklist("extract", {
  classify: {
    description: "Determine the document type",
    outputSchema: { type: "object", required: ["category", "confidence"], properties: { category: { type: "string" }, confidence: { type: "number" } } },
  },
  load_guidance: {
    description: "Load extraction guidance from knowledge",
    dependsOn: ["classify"],
    outputSchema: { type: "object", required: ["guidanceLoaded"], properties: { guidanceLoaded: { type: "boolean" }, category: { type: "string" } } },
  },
  extract: {
    description: "Extract structured data from the image",
    dependsOn: ["load_guidance"],
    outputSchema: { type: "object", required: ["rawExtraction"], properties: { rawExtraction: { type: "object" } } },
  },
  validate: {
    description: "Validate extracted data",
    dependsOn: ["extract"],
    outputSchema: { type: "object", required: ["validatedData", "issues"], properties: { validatedData: { type: "object" }, issues: { type: "array", items: { type: "string" } } } },
  },
  save: {
    description: "Save to health profile",
    dependsOn: ["validate"],
    outputSchema: { type: "object", required: ["saved"], properties: { saved: { type: "boolean" }, recordId: { type: "string" } } },
  },
  submit: {
    description: "Submit the extraction result",
    dependsOn: ["save"],
  },
});

// ── classify ──────────────────────────────────────────────────────
__flow.start("classify");
const classification = await analyzeImage(
  base64Data,
  "Classify this medical document into one of: lab_results, prescription, doctor_notes, medication_label, imaging_report, vaccination, allergy, surgical_report, discharge_summary, referral, vital_signs. Return JSON: { \"category\": string, \"confidence\": number }"
) as AnalyzeImageResult;
const { category, confidence } = JSON.parse(classification.text) as { category: string; confidence: number };
__flow.finish("classify", { category, confidence });

// ── load_guidance ─────────────────────────────────────────────────
__flow.start("load_guidance");
Space.current().loadKnowledge("document_type", "category", category);
await inspect();
__flow.finish("load_guidance", { guidanceLoaded: true, category });

// ── extract ───────────────────────────────────────────────────────
__flow.start("extract");
// Use the extraction prompt from the loaded knowledge
const extraction = await analyzeImage(
  base64Data,
  `Extract all structured data from this ${category} document. Return as JSON with all relevant fields.`
) as AnalyzeImageResult;
const rawExtraction = JSON.parse(extraction.text) as Record<string, unknown>;
__flow.finish("extract", { rawExtraction });

// ── yield ─────────────────────────────────────────────────────────
await inspect({ category, confidence, rawExtraction });
```

End the cycle here. Cycle 2 will validate, save, and submit.
