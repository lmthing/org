# Phase 2 — Validate and Save

Validate the extracted data, save to profile, and submit.

```ts
// ── validate ──────────────────────────────────────────────────────
__flow.start("validate");
const issues: string[] = [];

// Check required fields based on category
if (!rawExtraction) {
  issues.push("No extraction data available");
} else {
  // Category-specific validation
  if (category === "lab_results") {
    if (!rawExtraction.date) issues.push("Missing date");
    if (!rawExtraction.values || !Array.isArray(rawExtraction.values)) issues.push("Missing lab values array");
  }
  // Add more category-specific checks as needed
}

const validatedData = { ...rawExtraction, category, validationIssues: issues.length > 0 ? issues : undefined };
__flow.finish("validate", { validatedData, issues });

// ── save ──────────────────────────────────────────────────────────
__flow.start("save");
checkpoint("before-save");
const saveResult = await saveRecord(category, validatedData as Record<string, unknown>) as SaveRecordResult;

// Display the extraction result
if (category === "lab_results" && rawExtraction.values) {
  display(<LabResultCard
    panelName={(rawExtraction.panelName as string) ?? "Lab Results"}
    date={(rawExtraction.date as string) ?? new Date().toISOString()}
    values={(rawExtraction.values as LabValue[]) ?? []}
    labName={rawExtraction.labName as string}
  />);
}
__flow.finish("save", { saved: saveResult.saved, recordId: saveResult.recordId });

// ── submit ────────────────────────────────────────────────────────
__flow.start("submit");
submitExtraction({ category, data: validatedData, saved: true });
```
