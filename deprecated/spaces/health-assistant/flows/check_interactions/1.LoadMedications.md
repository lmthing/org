# Phase 1 — Load, Categorize, and Check

Load medications from the profile, categorize by drug class, and delegate interaction checks.

```ts
const __flow = tasklist("check_interactions", {
  load_medications: {
    description: "Load medications and conditions",
    outputSchema: { type: "object", required: ["medications", "conditions"], properties: { medications: { type: "array" }, conditions: { type: "array" } } },
  },
  categorize: {
    description: "Categorize medications by class",
    dependsOn: ["load_medications"],
    outputSchema: { type: "object", required: ["drugClasses", "pairsToCheck"], properties: { drugClasses: { type: "array" }, pairsToCheck: { type: "array" } } },
  },
  check_pairs: {
    description: "Check interactions for each pair",
    dependsOn: ["categorize"],
    outputSchema: { type: "object", required: ["interactions"], properties: { interactions: { type: "array" } } },
  },
  compile_report: {
    description: "Compile interaction report",
    dependsOn: ["check_pairs"],
    outputSchema: { type: "object", required: ["report"], properties: { report: { type: "object" } } },
  },
  submit: {
    description: "Submit the report",
    dependsOn: ["compile_report"],
  },
});

// ── load_medications ──────────────────────────────────────────────
__flow.start("load_medications");
const history = await loadHistory() as HealthHistory;
const medications = history.categories.medications ?? history.categories.prescription ?? [];
const conditions = history.categories.doctor_notes ?? history.categories.discharge_summary ?? [];
__flow.finish("load_medications", { medications, conditions });

// ── categorize ────────────────────────────────────────────────────
__flow.start("categorize");
// Build all pairs of medications to check
const pairsToCheck: Array<{ drug1: string; drug2?: string; type: string }> = [];
// Drug-drug pairs
for (let i = 0; i < medications.length; i++) {
  for (let j = i + 1; j < medications.length; j++) {
    pairsToCheck.push({ drug1: medications[i].name as string, drug2: medications[j].name as string, type: "drug_drug" });
  }
  // Drug-condition pairs
  for (const cond of conditions) {
    pairsToCheck.push({ drug1: medications[i].name as string, drug2: cond.diagnosis as string, type: "drug_condition" });
  }
}
__flow.finish("categorize", { drugClasses: [], pairsToCheck });

// ── check_pairs ───────────────────────────────────────────────────
__flow.start("check_pairs");
const interactions: Array<{ pair: string; severity: string; mechanism: string; recommendation: string }> = [];
// Load interaction knowledge for the first type
Space.current().loadKnowledge("interaction", "type", "drug_drug");
await inspect();

// Delegate research for high-priority pairs
for (const pair of pairsToCheck.slice(0, 10)) {
  const result = await delegate({
    space: "health-assistant",
    agent: "researcher",
    flow: "investigate",
    task: `Check ${pair.type} interaction between ${pair.drug1} and ${pair.drug2}. Return: interaction severity (minor/moderate/severe/contraindicated), mechanism, recommendation.`,
  }) as { output: string; status: "ok" | "error" };
  // Parse result and add to interactions array
}
__flow.finish("check_pairs", { interactions });

// ── yield ─────────────────────────────────────────────────────────
await inspect({ medications, conditions, interactions });
```
