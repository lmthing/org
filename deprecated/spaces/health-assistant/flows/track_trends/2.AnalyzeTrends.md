# Phase 2 — Generate Report and Submit

Compose the trend report and submit.

```ts
// ── generate_report ───────────────────────────────────────────────
__flow.start("generate_report");
const improving = trends.filter(t => t.direction === "improving");
const declining = trends.filter(t => t.direction === "declining");
const stable = trends.filter(t => t.direction === "stable");

const report = {
  totalMetricsTracked: trends.length,
  improving: improving.map(t => t.name),
  declining: declining.map(t => t.name),
  stable: stable.map(t => t.name),
  dateRange: history.dateRange,
  summary: `Tracking ${trends.length} metrics from ${history.dateRange.earliest ?? "N/A"} to ${history.dateRange.latest ?? "N/A"}. ${improving.length} improving, ${declining.length} declining, ${stable.length} stable.`,
  flaggedConcerns,
};
__flow.finish("generate_report", { report });

// ── submit ────────────────────────────────────────────────────────
__flow.start("submit");
submitTrendReport({
  metrics: trends,
  summary: report.summary,
  flaggedConcerns,
  disclaimer: "This trend analysis is for informational purposes only. It does not constitute medical advice. Consult a healthcare professional for interpretation of lab results and health trends.",
});
```
