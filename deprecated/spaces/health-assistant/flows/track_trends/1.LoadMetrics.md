# Phase 1 — Load and Analyze

Load all time-series health data and analyze trends.

```ts
const __flow = tasklist("track_trends", {
  load_metrics: {
    description: "Load all time-series data",
    outputSchema: { type: "object", required: ["metrics", "dateRange"], properties: { metrics: { type: "array" }, dateRange: { type: "object" } } },
  },
  analyze_trends: {
    description: "Analyze trends for each metric",
    dependsOn: ["load_metrics"],
    outputSchema: { type: "object", required: ["trends", "flaggedConcerns"], properties: { trends: { type: "array" }, flaggedConcerns: { type: "array", items: { type: "string" } } } },
  },
  generate_report: {
    description: "Generate visual trend report",
    dependsOn: ["analyze_trends"],
    outputSchema: { type: "object", required: ["report"], properties: { report: { type: "object" } } },
  },
  submit: {
    description: "Submit the report",
    dependsOn: ["generate_report"],
  },
});

// ── load_metrics ──────────────────────────────────────────────────
__flow.start("load_metrics");
const history = await loadHistory() as HealthHistory;
pin("history");

// Extract all dated records as time-series data points
const metrics: Array<{ date: string; category: string; name: string; value: unknown; unit?: string; referenceRange?: string }> = [];
for (const [category, records] of Object.entries(history.categories)) {
  for (const record of records) {
    if (category === "lab_results" && Array.isArray(record.values)) {
      for (const v of record.values as Array<Record<string, unknown>>) {
        metrics.push({
          date: record.date as string,
          category,
          name: v.testName as string,
          value: v.value,
          unit: v.unit as string,
          referenceRange: v.referenceRange as string,
        });
      }
    } else if (category === "vital_signs") {
      metrics.push({
        date: record.date as string,
        category,
        name: record.type as string ?? category,
        value: record.value,
        unit: record.unit as string,
      });
    }
  }
}

const dateRange = history.dateRange;
__flow.finish("load_metrics", { metrics, dateRange });

// ── analyze_trends ────────────────────────────────────────────────
__flow.start("analyze_trends");

// Group metrics by name and analyze trends
const byName = new Map<string, Array<{ date: string; value: unknown }>>();
for (const m of metrics) {
  if (!byName.has(m.name)) byName.set(m.name, []);
  byName.get(m.name)!.push({ date: m.date, value: m.value });
}

const trends: Array<{ name: string; direction: string; dataPoints: number; latest: unknown; referenceRange?: string }> = [];
const flaggedConcerns: string[] = [];

for (const [name, points] of byName) {
  const sorted = points.sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime());
  const direction = sorted.length >= 2 ? "analyzed" : "insufficient data";
  trends.push({ name, direction, dataPoints: sorted.length, latest: sorted.at(-1)?.value });
}

// Load specialty knowledge for reference range context
Space.current().loadKnowledge("specialty", "area", "general");
await inspect();

__flow.finish("analyze_trends", { trends, flaggedConcerns });

// ── yield ─────────────────────────────────────────────────────────
display(<HealthTimeline events={
  metrics.map(m => ({ date: m.date, type: m.category, title: `${m.name}: ${m.value}${m.unit ? ` ${m.unit}` : ""}`, details: m }))
} />);
await inspect({ metrics, trends, flaggedConcerns });
```
