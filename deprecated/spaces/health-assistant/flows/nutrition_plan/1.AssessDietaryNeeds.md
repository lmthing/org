# Phase 1 — Assess Dietary Needs

Load profile, identify restrictions, and research guidelines.

```ts
const __flow = tasklist("nutrition_plan", {
  assess: {
    description: "Load profile and identify dietary factors",
    outputSchema: { type: "object", required: ["conditions", "medications", "recentLabs"], properties: { conditions: { type: "array" }, medications: { type: "array" }, recentLabs: { type: "array" } } },
  },
  identify_restrictions: {
    description: "Identify dietary restrictions",
    dependsOn: ["assess"],
    outputSchema: { type: "object", required: ["restrictions", "supplementWarnings"], properties: { restrictions: { type: "array" }, supplementWarnings: { type: "array" } } },
  },
  research_guidelines: {
    description: "Research nutrition guidelines",
    dependsOn: ["identify_restrictions"],
    optional: true,
    outputSchema: { type: "object", properties: { researchResults: { type: "array" } } },
  },
  compose_plan: {
    description: "Compose nutrition plan",
    dependsOn: ["research_guidelines"],
    outputSchema: { type: "object", required: ["plan"], properties: { plan: { type: "object" } } },
  },
  submit: {
    description: "Submit the plan",
    dependsOn: ["compose_plan"],
  },
});

// ── assess ────────────────────────────────────────────────────────
__flow.start("assess");
const history = await loadHistory() as HealthHistory;
const conditions = history.categories.doctor_notes ?? history.categories.discharge_summary ?? [];
const medications = history.categories.medications ?? history.categories.prescription ?? [];
const recentLabs = history.categories.lab_results ?? [];
__flow.finish("assess", { conditions, medications, recentLabs });

// ── identify_restrictions ─────────────────────────────────────────
__flow.start("identify_restrictions");
const restrictions: string[] = [];
const supplementWarnings: string[] = [];

// Load diet knowledge per condition
// Example: if diabetes found, load diabetic diet
Space.current().loadKnowledge("nutrition", "diet", "Mediterranean");
Space.current().loadKnowledge("nutrition", "supplement", "vitamins");
await inspect();

// Cross-reference medications with supplement interactions
__flow.finish("identify_restrictions", { restrictions, supplementWarnings });

// ── research_guidelines ───────────────────────────────────────────
__flow.start("research_guidelines");
const researchResults: Array<{ topic: string; output: string }> = [];
// Delegate nutrition research
for (const condition of conditions.slice(0, 3)) {
  const result = await delegate({
    space: "health-assistant",
    agent: "researcher",
    flow: "investigate",
    task: `Research evidence-based dietary guidelines for: ${condition.diagnosis ?? condition.title}. Include specific foods to include and avoid.`,
  }) as { output: string; status: "ok" | "error" };
  researchResults.push({ topic: condition.diagnosis as string, output: result.output });
}
__flow.finish("research_guidelines", { researchResults });

// ── yield ─────────────────────────────────────────────────────────
await inspect({ conditions, medications, restrictions, supplementWarnings, researchResults });
```
