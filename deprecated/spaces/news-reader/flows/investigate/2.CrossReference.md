# Phase 2 — Extract and cross-reference

Extract verifiable claims and entities, then cross-reference across sources.

```ts
// ── extract ──────────────────────────────────────────────────────────────
__flow.start("extract");
const excerpts = /* from read */;
const claims: Array<{
  claim: string;
  sourceId: string;
  confidence: "confirmed" | "likely" | "unverified" | "disputed";
  evidence: string;
}> = [];
const entities: Array<{ text: string; type: string; count: number }> = [];

for (const exc of excerpts) {
  if (!exc.text) continue;
  const excEntities = await extractEntities(exc.text, {
    types: ["person", "organisation", "location", "date", "money"],
    maxEntities: 20,
  });
  entities.push(...excEntities.map(e => ({ text: e.text, type: e.type, count: e.count })));
  // Extract atomic factual claims from the text, each bound to its sourceId
  // A claim is a single factual statement that can be verified independently
}
__flow.finish("extract", { claims, entities });

// ── cross_reference ──────────────────────────────────────────────────────
__flow.start("cross_reference");
// Group claims by similarity. A claim confirmed by 2+ sources → "confirmed".
// A claim appearing in only 1 source → "unverified".
// Claims that contradict each other → "disputed".
const confirmed: string[] = [];
const disputed: Array<{ claim: string; positions: Array<{ sourceId: string; stance: string }> }> = [];
const unverified: string[] = [];
__flow.finish("cross_reference", { confirmed, disputed, unverified });

await inspect(claims, entities, { confirmed, disputed, unverified });
```
