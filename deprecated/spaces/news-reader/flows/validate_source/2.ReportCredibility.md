# Phase 2 — Corroborate and assess

Search for corroborating coverage, then produce the final assessment.

```ts
// ── corroborate ──────────────────────────────────────────────────────────
__flow.start("corroborate");
const claims = /* from analyze */;
const corroboration: any[] = [];
const contradictions: any[] = [];

// For each major claim, search for coverage from high-credibility sources
for (const claim of claims.slice(0, 5)) {
  try {
    const results = await searchNews(claim.claim, {
      freshness: "month",
      topK: 5,
      domains: ["reuters.com", "apnews.com", "bbc.com"],
    });
    if (results.length > 0) {
      corroboration.push({ claim: claim.claim, sources: results.map(r => r.source) });
    }
  } catch { /* skip */ }
}
__flow.finish("corroborate", { corroboration, contradictions });

// ── assess ───────────────────────────────────────────────────────────────
__flow.start("assess");
const info = /* domainInfo from analyze */;
const concerns: string[] = [];
let recommendation = "Trusted";

if (info.credibilityScore < 0.5) { concerns.push("Low credibility score"); recommendation = "Avoid"; }
else if (info.credibilityScore < 0.7) { concerns.push("Moderate credibility — verify claims independently"); recommendation = "Caution"; }
if (info.factualityRating === "low" || info.factualityRating === "very_low") { concerns.push("Poor factuality track record"); recommendation = "Avoid"; }
if (info.biasRating !== "center" && info.biasRating !== "unknown") { concerns.push(`Bias rating: ${info.biasRating} — consider cross-referencing`); }

const assessment = `## Source: ${info.domain}\nCredibility: ${info.credibilityScore}/1.0 | Bias: ${info.biasRating} | Factuality: ${info.factualityRating}`;

submitValidation({
  domain: info.domain,
  credibilityScore: info.credibilityScore,
  biasRating: info.biasRating,
  factualityRating: info.factualityRating,
  assessment,
  concerns,
  recommendation,
});
__flow.finish("assess");
```
