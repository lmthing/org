# Phase 1 — Search and validate

```ts
const __flow = tasklist("search_news", { /* DAG from harness */ });

// ── search ───────────────────────────────────────────────────────────────
__flow.start("search");
const articles = await searchNews(/* user query */, {
  freshness: "week",
  topK: 15,
});
if (articles.length === 0) {
  __flow.finish("search", { articles: [], query: /* user query */ });
} else {
  __flow.finish("search", { articles, query: /* user query */ });
}

// ── validate ─────────────────────────────────────────────────────────────
__flow.start("validate");
const rawArticles = /* from search */;
const validated = [];
for (const art of rawArticles) {
  try {
    const domain = new URL(art.url).hostname.replace(/^www\./, "");
    const info = await getDomainInfo(domain);
    validated.push({ ...art, credibility: info.credibilityScore, bias: info.biasRating, factuality: info.factualityRating });
  } catch {
    validated.push({ ...art, credibility: 0.5, bias: "unknown", factuality: "mixed" });
  }
}
validated.sort((a, b) => (b.credibility ?? 0) - (a.credibility ?? 0));
__flow.finish("validate", { articles: validated });

await inspect(validated);
```
