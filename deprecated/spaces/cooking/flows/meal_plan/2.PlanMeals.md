# Phase 2 — Propose meals in parallel and review

```ts
// ── propose_meals ────────────────────────────────────────────────────────
__flow.start("propose_meals");

const days = ["Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday", "Sunday"];

// Fork one worker per day — each independently proposes meals for that day.
// XS model is enough: it's a structured generation task with a narrow output schema.
const dayForks = days.map(day =>
  fork<DayMeals>({
    instruction: [
      `[model:XS] Propose meals for ${day}. Constraints:`,
      `  - People: ${__people}`,
      `  - Meal types: ${JSON.stringify(__mealTypes)}`,
      `  - Restrictions: ${JSON.stringify(__restrictions ?? [])}`,
      `  - Cuisine preferences: ${JSON.stringify(__cuisinePrefs ?? [])}`,
      `  - Max cook time: ${__maxMinutesPerDay} minutes`,
      `  - Use-soon pantry items: ${JSON.stringify((__pantryData as any)?.useSoonItems ?? [])}`,
      `Return DayMeals: { day: string; meals: Array<{ type: string; dish: string; estimatedMinutes: number; mainIngredients: string[] }> }`,
    ].join("\n"),
    tokenBudget: 2000,
  })
);

display(<Markdown>{"Planning meals for all 7 days in parallel…"}</Markdown>, { id: "plan-progress" });
await inspect(...dayForks);

const meals = (dayForks as unknown as DayMeals[]).flatMap(d => (d?.meals ?? []).map(m => ({ ...m, day: d.day })));

display(
  <Table data={meals.map(m => ({ Day: m.day, Type: m.type, Dish: m.dish, "~mins": m.estimatedMinutes }))} />,
  { id: "meal-table" },
);

__flow.finish("propose_meals", { meals });

// ── review_plan ──────────────────────────────────────────────────────────
__flow.start("review_plan");

const approvalAnswer = ask<string>(
  <Confirm message="Does this meal plan look good? (Say 'yes' to confirm, or describe what to swap)" />,
  { fallback: "yes", timeout: 300000 },
);
await inspect(approvalAnswer);

let approvedMeals = meals;
const resp = (approvalAnswer as string).toLowerCase();

if (resp !== "yes" && resp !== "y") {
  // User wants swaps — use M_R for reasoning about which meals to change
  const swapFork = fork<DayMeals[]>({
    instruction: `[model:M_R] The user wants changes to the meal plan: "${approvalAnswer}". Current plan: ${JSON.stringify(meals)}. Constraints: ${JSON.stringify({ restrictions: __restrictions, cuisinePrefs: __cuisinePrefs, maxMinutesPerDay: __maxMinutesPerDay })}. Apply the requested changes and return the full updated plan as DayMeals[].`,
    tokenBudget: 8000,
  });
  await inspect(swapFork);
  approvedMeals = (swapFork as unknown as DayMeals[]).flatMap(d => (d?.meals ?? []).map(m => ({ ...m, day: d.day })));

  display(
    <Table data={approvedMeals.map(m => ({ Day: m.day, Type: m.type, Dish: m.dish, "~mins": m.estimatedMinutes }))} />,
    { id: "meal-table" },
  );
}

__flow.finish("review_plan", { approvedMeals });
pin("approvedMeals");
checkpoint("after-meal-review");

await inspect(approvedMeals);
```
