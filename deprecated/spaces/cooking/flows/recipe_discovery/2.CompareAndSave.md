# Phase 2 — Compare and save the best recipe

`__rawRecipes` holds the parsed recipes from phase 1. Complete both tasks and call the sink.

```ts
const __flow = tasklist("recipe_discovery", { /* DAG from harness */ });

// ── compare ──────────────────────────────────────────────────────────────
__flow.start("compare");

const recipes = Array.isArray(__rawRecipes) ? __rawRecipes as object[] : [];

// Rank by simplicity (fewer ingredients) and technique clarity
const ranked = recipes.map((r, i) => ({
  rank: i + 1,
  score: Math.max(1, 10 - ((r as { ingredients?: string[] }).ingredients?.length ?? 5)),
  title: (r as { title?: string }).title ?? `Recipe ${i + 1}`,
  source: (r as { source?: string }).source ?? "unknown",
  recipe: r,
})).sort((a, b) => b.score - a.score).map((r, i) => ({ ...r, rank: i + 1 }));

__flow.finish("compare", { rankedRecipes: ranked });

// ── select_and_save ───────────────────────────────────────────────────────
__flow.start("select_and_save");

const best = ranked[0];
const selectedRecipe = best?.recipe ?? {};

try {
  const parsed = await parseIngredients(
    ((selectedRecipe as { ingredients?: string[] }).ingredients ?? []).join("\n")
  );
  const normalized = { ...(selectedRecipe as object), ingredients: parsed };
  const slug = ((selectedRecipe as { title?: string }).title ?? "recipe")
    .toLowerCase().replace(/\s+/g, "-").replace(/[^a-z0-9-]/g, "");
  Space.current().write(`recipes/${slug}.json`, JSON.stringify(normalized, null, 2));
  __flow.finish("select_and_save", { selectedRecipe: normalized });
  submitDiscoveredRecipe(normalized);
} catch {
  __flow.finish("select_and_save", { selectedRecipe });
  submitDiscoveredRecipe(selectedRecipe);
}
```
