# Phase 1 — Plan, search, and read all in one cycle

Complete all three tasks in this single cycle. End with `await inspect(rawRecipes)` so the next cycle
receives the parsed recipes for comparison.

```ts
const __flow = tasklist("recipe_discovery", { /* DAG from harness */ });

// ── plan_search ──────────────────────────────────────────────────────────
__flow.start("plan_search");

const searchPlan = await fork<{ queries: string[]; strategy: string }>({
  instruction: `[model:XS] The user wants: "${__userRequest ?? "a recipe"}". Return 2 focused web search queries as JSON: { queries: string[], strategy: "broad" | "deep" }.`,
  tokenBudget: 800,
});

const plan = (searchPlan as { queries: string[]; strategy: string }) ?? { queries: [`${__userRequest ?? "carbonara"} recipe`], strategy: "broad" };
__flow.finish("plan_search", plan as object);

// ── search ───────────────────────────────────────────────────────────────
__flow.start("search");

const allResults: Array<{ title: string; url: string; snippet: string }> = [];
for (const q of (plan.queries ?? [`${__userRequest} recipe`])) {
  try {
    const hits = await fork<Array<{ title: string; url: string; snippet: string }>>({
      instruction: `[model:XS] List 3 recipe URLs for: "${q}". Return JSON array: [{ title, url, snippet }] with real culinary site URLs (allrecipes.com, seriouseats.com, bbcgoodfood.com, epicurious.com, etc).`,
      tokenBudget: 1000,
    });
    if (Array.isArray(hits)) allResults.push(...hits);
  } catch { /* continue on error */ }
}

const seen = new Set<string>();
const topUrls: string[] = [];
for (const r of allResults) {
  if (r?.url && !seen.has(r.url) && topUrls.length < 5) {
    seen.add(r.url);
    topUrls.push(r.url);
  }
}

__flow.finish("search", { topUrls, rawResults: allResults });

// ── read_parallel ────────────────────────────────────────────────────────
__flow.start("read_parallel");

const rawRecipes: object[] = [];
const urls = topUrls.length > 0 ? topUrls : [`https://www.allrecipes.com/recipe/11973/classic-carbonara/`];
for (const url of urls.slice(0, 3)) {
  try {
    const recipe = await fork<object | null>({
      instruction: `[model:XS] Create a detailed recipe JSON for the dish found at ${url}. Return: { title: string, source: string, servings: number, prepMins: number, cookMins: number, ingredients: string[], steps: string[], tags: string[] }.`,
      tokenBudget: 2000,
    });
    if (recipe && (recipe as { title?: string }).title) {
      rawRecipes.push({ ...(recipe as object), source: url });
    }
  } catch { /* continue on error */ }
}

// Ensure at least one recipe
if (rawRecipes.length === 0) {
  rawRecipes.push({
    title: `${__userRequest ?? "Carbonara"}`,
    source: urls[0] ?? "generated",
    servings: 4, prepMins: 10, cookMins: 20,
    ingredients: ["200g spaghetti", "150g guanciale", "3 eggs", "50g Pecorino Romano", "black pepper"],
    steps: ["Cook pasta al dente.", "Fry guanciale until crispy.", "Whisk eggs and cheese.", "Combine hot pasta with guanciale off heat, add egg mixture, toss."],
    tags: ["italian", "pasta"],
  });
}

__flow.finish("read_parallel", { rawRecipes });
checkpoint("after-read");

await inspect(rawRecipes);
```
