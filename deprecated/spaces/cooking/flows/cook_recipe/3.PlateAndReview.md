# Phase 3 — Plate and report

```ts
// ── plate ────────────────────────────────────────────────────────────────
__flow.start("plate");

// Load cuisine knowledge for plating guidance
Space.current().loadKnowledge("cuisine", "style");
await inspect();

const cuisineTags = ((__recipe as Recipe).tags ?? []).map(t => t.toLowerCase());

// Fork plating advice — XS is enough for a short tips list
const platingFork = fork<string[]>({
  instruction: `[model:XS] Give 3 concise plating tips for "${(__recipe as Recipe).title}". Consider cuisine style: ${cuisineTags.join(", ") || "general"}. Return string[].`,
  tokenBudget: 1500,
});
await inspect(platingFork);

const tips = platingFork as unknown as string[];
display(
  <Markdown>{"## Plating Tips\n\n" + tips.map(t => `- ${t}`).join("\n")}</Markdown>,
  { id: "plating-tips" },
);

const rating = ask<string>(
  <Select options={["5 - Perfect", "4 - Great", "3 - Good", "2 - Needs work", "1 - Try again"]}
    label="How did it turn out?" />,
  { fallback: "4 - Great" },
);
const feedback = ask<string>(
  <TextInput label="Any feedback or notes for next time?" />,
  { fallback: "" },
);
await inspect(rating, feedback);

const ratingNum = parseInt((rating as string).charAt(0), 10);
__flow.finish("plate", { rating: ratingNum, feedback: feedback as string });

// ── report ───────────────────────────────────────────────────────────────
__flow.start("report");

const report = {
  dish: (__recipe as Recipe).title,
  outcome: ratingNum >= 4 ? "success" : ratingNum === 3 ? "acceptable" : "needs-retry",
  notes: [
    feedback as string,
    ...(__adjustments ?? []),
    ...(((__nutritionResult as any)?.warnings ?? []) as string[]),
  ].filter(Boolean).join(" | "),
};

// Save an updated recipe if the user made adjustments worth keeping
if ((__adjustments as string[])?.length > 0 && ratingNum >= 4) {
  checkpoint("before-recipe-update");
  const updatedRecipe = {
    ...((__recipe as Recipe)),
    notes: report.notes,
  };
  Space.current().write(
    `recipes/${(__recipe as Recipe).title.toLowerCase().replace(/\s+/g, "-")}.json`,
    JSON.stringify(updatedRecipe, null, 2),
  );
}

__flow.finish("report");
submitCookingReport(report);
```

No `inspect()` in the final phase — call `submitCookingReport` directly. The session terminates as soon as the sink fires.
