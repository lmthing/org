# Phase 1 — Clarify, source recipe, and scale

```ts
const __flow = tasklist("cook_recipe", { /* DAG from harness */ });

// ── clarify ──────────────────────────────────────────────────────────────
__flow.start("clarify");

// Ask for dish details if not already in scope
const dishName = ask<string>(
  <TextInput label="What would you like to cook today?" />,
  { fallback: "pasta carbonara" },
);
const servingsAnswer = ask<string>(
  <TextInput label="How many servings?" placeholder="e.g. 4" />,
  { fallback: "4" },
);
const restrictionsAnswer = ask<string>(
  <TextInput label="Any dietary restrictions? (leave blank if none)" />,
  { fallback: "" },
);

await inspect(dishName, servingsAnswer, restrictionsAnswer);

// next cycle: dishName, servingsAnswer, restrictionsAnswer are resolved strings
const dish = dishName as string;
const servings = parseInt(servingsAnswer as string, 10) || 4;
const restrictions = (restrictionsAnswer as string)
  .split(",").map(s => s.trim()).filter(Boolean);

__flow.finish("clarify", { dish, servings, restrictions });

// ── source_recipe ────────────────────────────────────────────────────────
__flow.start("source_recipe");

// Check session space for a saved recipe first
const spaceFiles = Space.current().list("recipes");
const savedName = dish.toLowerCase().replace(/\s+/g, "-") + ".json";
let recipe: Recipe | null = null;

if (spaceFiles.includes(savedName)) {
  recipe = JSON.parse(Space.current().read("recipes/" + savedName)) as Recipe;
  display(<Markdown>{`Found saved recipe: **${recipe.title}**`}</Markdown>);
} else {
  display(<Markdown>{`No saved recipe found for **${dish}** — searching online…`}</Markdown>);

  // Always await fork() directly — the result is the value, not a future
  recipe = await fork<Recipe | null>({
    instruction: `[model:M] Find an authentic recipe for "${dish}". Return ONLY a JSON object matching this exact schema (no markdown, no prose): { "title": string, "servings": number, "ingredients": [{ "ingredient": string, "quantity": number, "unit": string }], "steps": string[] }. Use realistic quantities. Return null if the dish is unknown.`,
    tokenBudget: 4000,
  });
}

if (!recipe) {
  display(<Markdown>{"**Recipe not found.** Please try a different dish name or provide a recipe directly."}</Markdown>);
  __flow.finish("source_recipe", { recipe: null });
} else {
  __flow.finish("source_recipe", { recipe });
}

// ── scale ────────────────────────────────────────────────────────────────
__flow.start("scale");

const scaledIngredients = recipe
  ? await scaleRecipe(recipe.ingredients, servings / (recipe.servings || servings)) as ScaledIngredient[]
  : [];

// Display scaled ingredient list
display(
  <Table data={scaledIngredients.map(i => ({ ingredient: i.ingredient, quantity: `${i.quantity} ${i.unit}` }))} />,
  { id: "ingredients" }
);

__flow.finish("scale", { scaledIngredients });

// Pin key state — it must survive context compaction through the long cook cycle
pin("recipe");
pin("scaledIngredients");
checkpoint("after-scale");

await inspect(recipe, scaledIngredients);
```
