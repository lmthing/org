# Phase 2 — Nutrition check (parallel), prep, and cook

```ts
// ── nutrition_check (parallel, optional) ────────────────────────────────
__flow.start("nutrition_check");

// Run nutrition analysis in a fork — doesn't block the prep chain
const nutritionFork = fork<{ nutritionSummary: object; fitCheckPassed: boolean; warnings: string[] }>({
  instruction: [
    "[model:M] Analyse the nutritional content of these scaled ingredients: ",
    JSON.stringify(__scaledIngredients),
    ". Check against restrictions: ", JSON.stringify(__restrictions ?? []),
    ". Return { nutritionSummary, fitCheckPassed, warnings }.",
  ].join(""),
  tokenBudget: 6000,
  exclude: ["recipe"], // already pinned; no need to copy to fork
});
// Don't block on this — pass to inspect at the end of the phase

// ── prep ─────────────────────────────────────────────────────────────────
__flow.start("prep");

const prepSteps = (__recipe as Recipe).steps.filter(
  s => /chop|dice|mince|slice|peel|marinate|measure|mise|preheat|wash|trim/i.test(s)
);

display(<Markdown>{"## Mise en Place\n\nGather and prepare the following before you start cooking:"}</Markdown>);
display(<Table data={(__scaledIngredients as ScaledIngredient[]).map(i => ({ ingredient: i.ingredient, amount: `${i.quantity} ${i.unit}` }))} />);

// Walk the user through each prep step with confirmation
let userNotes = "";
for (let i = 0; i < prepSteps.length; i++) {
  const step = prepSteps[i]!;
  display(<Markdown>{`**Prep step ${i + 1}:** ${step}`}</Markdown>, { id: `prep-${i}` });
  const confirmed = ask<string>(
    <Confirm message={`Ready for next prep step? (step ${i + 2} of ${prepSteps.length})`} />,
    { fallback: "yes", timeout: 300000 },
  );
  await inspect(confirmed);
}

const prepNotes = ask<string>(
  <TextInput label="Any notes before cooking starts?" placeholder="e.g. doubled the garlic" />,
  { fallback: "" },
);
await inspect(prepNotes);
userNotes = prepNotes as string;

__flow.finish("prep", { prepComplete: true, userNotes });

// ── cook ─────────────────────────────────────────────────────────────────
__flow.start("cook");
checkpoint("before-cook");

const cookSteps = (__recipe as Recipe).steps.filter(
  s => !/chop|dice|mince|slice|peel|marinate|measure|mise|preheat|wash|trim/i.test(s)
);
const adjustments: string[] = [];

for (let i = 0; i < cookSteps.length; i++) {
  const step = cookSteps[i]!;
  display(<Markdown>{`**Step ${i + 1} / ${cookSteps.length}:** ${step}`}</Markdown>, { id: `cook-step-${i}` });

  const stepConfirm = ask<string>(
    <Confirm message={`Step ${i + 1} done? (or describe any adjustment)`} />,
    { fallback: "done", timeout: 600000 },
  );
  await inspect(stepConfirm);

  const resp = stepConfirm as string;
  if (resp && resp !== "done" && resp !== "yes") {
    adjustments.push(`Step ${i + 1}: ${resp}`);
  }
}

__flow.finish("cook", { cookComplete: true, adjustments });

// Resolve nutrition fork now that cooking is done
await inspect(nutritionFork);
const nutritionResult = nutritionFork as unknown as { nutritionSummary: object; fitCheckPassed: boolean; warnings: string[] };
__flow.finish("nutrition_check", nutritionResult);

if ((nutritionResult?.warnings ?? []).length > 0) {
  display(
    <Markdown>{"⚠ **Nutrition warnings:**\n" + nutritionResult.warnings.join("\n- ")}</Markdown>,
    { id: "nutrition-warnings" },
  );
}

await inspect(nutritionResult, adjustments);
```
