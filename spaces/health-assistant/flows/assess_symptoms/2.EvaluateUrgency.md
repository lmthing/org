# Phase 2 — Recommend and Submit

Determine care setting and submit the triage assessment.

```ts
// ── recommend ─────────────────────────────────────────────────────
__flow.start("recommend");
const careSettingMap: Record<string, string> = {
  emergency: "Emergency Room — Call 911 or go to nearest ER immediately",
  urgent: "Urgent Care — Seek care within 24 hours",
  semi_urgent: "Primary Care — Schedule appointment within 1-2 weeks",
  routine: "Self-Care / Telehealth — Monitor and consult if symptoms worsen",
};
const careSetting = careSettingMap[urgency] ?? careSettingMap.routine;
const recommendation = `${careSetting}. ${reasoning}`;
__flow.finish("recommend", { recommendation, careSetting });

// ── submit ────────────────────────────────────────────────────────
__flow.start("submit");
await saveRecord("triage_events", {
  symptoms,
  urgency,
  reasoning,
  recommendation,
  careSetting,
  severity,
} as Record<string, unknown>);

submitTriageAssessment({
  symptoms,
  urgency,
  reasoning,
  recommendation,
  disclaimer: "This is an informational assessment, not a medical diagnosis. If you believe you are experiencing a medical emergency, call emergency services immediately.",
});
```
