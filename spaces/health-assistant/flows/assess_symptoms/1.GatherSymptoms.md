# Phase 1 — Gather, Contextualize, Evaluate

Collect symptom details, load relevant history, and evaluate urgency.

```ts
const __flow = tasklist("assess_symptoms", {
  gather: {
    description: "Gather symptom details",
    outputSchema: { type: "object", required: ["symptoms", "severity"], properties: { symptoms: { type: "string" }, severity: { type: "number" }, onset: { type: "string" }, duration: { type: "string" }, associatedSymptoms: { type: "array", items: { type: "string" } } } },
  },
  contextualize: {
    description: "Load relevant history",
    dependsOn: ["gather"],
    outputSchema: { type: "object", required: ["relevantHistory"], properties: { relevantHistory: { type: "object" } } },
  },
  evaluate: {
    description: "Evaluate urgency",
    dependsOn: ["contextualize"],
    outputSchema: { type: "object", required: ["urgency", "reasoning"], properties: { urgency: { type: "string" }, reasoning: { type: "string" } } },
  },
  recommend: {
    description: "Recommend care setting",
    dependsOn: ["evaluate"],
    outputSchema: { type: "object", required: ["recommendation", "careSetting"], properties: { recommendation: { type: "string" }, careSetting: { type: "string" } } },
  },
  submit: {
    description: "Submit assessment",
    dependsOn: ["recommend"],
  },
});

// ── gather ────────────────────────────────────────────────────────
__flow.start("gather");
// Use ask() to collect structured symptom information
const symptomInfo = await ask(<DocumentUpload prompt="Please describe your symptoms in detail. Include: onset, severity (1-10), duration, location, and any associated symptoms." />) as Record<string, unknown>;
const symptoms = (symptomInfo.symptoms ?? symptomInfo.description ?? "") as string;
const severity = (symptomInfo.severity ?? 5) as number;
__flow.finish("gather", {
  symptoms,
  severity,
  onset: symptomInfo.onset as string,
  duration: symptomInfo.duration as string,
  associatedSymptoms: (symptomInfo.associatedSymptoms ?? []) as string[],
});

// ── contextualize ─────────────────────────────────────────────────
__flow.start("contextualize");
const history = await loadHistory() as HealthHistory;
const relevantHistory = {
  conditions: history.categories.doctor_notes ?? [],
  medications: history.categories.medications ?? [],
  allergies: history.categories.allergy ?? [],
};
__flow.finish("contextualize", { relevantHistory });

// ── evaluate ──────────────────────────────────────────────────────
__flow.start("evaluate");
// Load urgency criteria based on symptom patterns
Space.current().loadKnowledge("triage", "urgency", "emergency");
await inspect();
// Apply red-flag criteria
// Determine urgency: emergency, urgent, semi_urgent, routine
const urgency = "routine"; // Will be determined by symptom analysis
const reasoning = "Based on symptom assessment and red-flag criteria evaluation.";
__flow.finish("evaluate", { urgency, reasoning });

// ── yield ─────────────────────────────────────────────────────────
await inspect({ symptoms, severity, urgency, reasoning });
```
