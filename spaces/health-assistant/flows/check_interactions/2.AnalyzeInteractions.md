# Phase 2 — Compile Report and Submit

Compile all found interactions into a severity-rated report and submit.

```ts
// ── compile_report ────────────────────────────────────────────────
__flow.start("compile_report");
const severeInteractions = interactions.filter(i => i.severity === "severe" || i.severity === "contraindicated");
const moderateInteractions = interactions.filter(i => i.severity === "moderate");
const minorInteractions = interactions.filter(i => i.severity === "minor");

const report = {
  totalInteractions: interactions.length,
  severe: severeInteractions,
  moderate: moderateInteractions,
  minor: minorInteractions,
  summary: `Found ${interactions.length} interactions: ${severeInteractions.length} severe, ${moderateInteractions.length} moderate, ${minorInteractions.length} minor.`,
};
__flow.finish("compile_report", { report });

// ── submit ────────────────────────────────────────────────────────
__flow.start("submit");
await saveRecord("interactions", report as Record<string, unknown>);
submitInteractionReport({
  medications: medications.map((m: Record<string, unknown>) => m.name as string),
  interactions,
  summary: report.summary,
  disclaimer: "This interaction check is for informational purposes only. Do not stop or change medications without consulting your healthcare provider.",
});
```
