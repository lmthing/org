# Phase 2 — Compose and Submit

Create the personalized nutrition plan and submit.

```ts
// ── compose_plan ──────────────────────────────────────────────────
__flow.start("compose_plan");
const plan = {
  conditions: conditions.map((c: Record<string, unknown>) => c.diagnosis ?? c.title),
  dietType: "Personalized",
  dailyTarget: {
    calories: "~2000 kcal (adjust based on individual needs)",
    protein: "20-25%",
    carbs: "40-45%",
    fat: "30-35%",
  },
  foods: {
    include: ["Vegetables", "Lean proteins", "Whole grains", "Fruits"],
    avoid: restrictions,
    limit: ["Processed foods", "Added sugars", "Excess sodium"],
  },
  supplements: supplementWarnings.length > 0
    ? supplementWarnings.map(w => ({ recommendation: w, caution: "Consult healthcare provider before starting" }))
    : [],
  mealPlan: "3 meals + 2 snacks daily, spaced 3-4 hours apart",
};
__flow.finish("compose_plan", { plan });

// ── submit ────────────────────────────────────────────────────────
__flow.start("submit");
await saveRecord("nutrition_plans", plan as Record<string, unknown>);
submitNutritionPlan({
  conditions: plan.conditions as string[],
  dietType: plan.dietType,
  dailyTarget: plan.dailyTarget,
  foods: plan.foods,
  supplements: plan.supplements as object[],
  mealPlan: plan.mealPlan as string,
  disclaimer: "This nutrition plan is for informational purposes only. Consult a registered dietitian or healthcare provider before making significant dietary changes.",
});
```
