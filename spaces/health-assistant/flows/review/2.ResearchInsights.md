# Phase 2 — Research Topics

Delegate research for identified health topics.

```ts
// ── research_topics ───────────────────────────────────────────────
__flow.start("research_topics");
const researchResults: Array<{ topic: string; output: string }> = [];

for (const topic of topics) {
  const result = await delegate({
    space: "health-assistant",
    agent: "researcher",
    flow: "investigate",
    task: topic,
  }) as { output: string; status: "ok" | "error" };

  researchResults.push({ topic, output: result.output });
}

__flow.finish("research_topics", { researchResults });
await inspect({ researchResults });
```
