# Phase 1 — Gather constraints and check pantry

```ts
const __flow = tasklist("meal_plan", { /* DAG from harness */ });

// ── gather_constraints ───────────────────────────────────────────────────
__flow.start("gather_constraints");

const peopleAnswer = ask<string>(
  <TextInput label="How many people are you planning for?" placeholder="e.g. 2" />,
  { fallback: "2" },
);
const mealTypesAnswer = ask<string>(
  <Select
    options={["Breakfast, Lunch & Dinner", "Lunch & Dinner", "Dinner only"]}
    label="Which meals should I plan?" />,
  { fallback: "Lunch & Dinner" },
);
const restrictionsAnswer = ask<string>(
  <TextInput label="Dietary restrictions (comma-separated, or blank)" placeholder="e.g. vegan, gluten-free" />,
  { fallback: "" },
);
const cuisineAnswer = ask<string>(
  <TextInput label="Cuisine preferences (comma-separated, or blank for variety)" placeholder="e.g. Italian, Asian" />,
  { fallback: "" },
);
const timeAnswer = ask<string>(
  <TextInput label="Max cooking time per day (minutes)?" placeholder="e.g. 45" />,
  { fallback: "45" },
);

await inspect(peopleAnswer, mealTypesAnswer, restrictionsAnswer, cuisineAnswer, timeAnswer);

const people = parseInt(peopleAnswer as string, 10) || 2;
const mealTypesMap: Record<string, string[]> = {
  "Breakfast, Lunch & Dinner": ["breakfast", "lunch", "dinner"],
  "Lunch & Dinner": ["lunch", "dinner"],
  "Dinner only": ["dinner"],
};
const mealTypes = mealTypesMap[mealTypesAnswer as string] ?? ["lunch", "dinner"];
const restrictions = (restrictionsAnswer as string).split(",").map(s => s.trim()).filter(Boolean);
const cuisinePrefs = (cuisineAnswer as string).split(",").map(s => s.trim()).filter(Boolean);
const maxMinutesPerDay = parseInt(timeAnswer as string, 10) || 45;

__flow.finish("gather_constraints", { people, mealTypes, restrictions, cuisinePrefs, maxMinutesPerDay });

// ── check_pantry ─────────────────────────────────────────────────────────
__flow.start("check_pantry");

// Load pantry-manager to check stock
const cooking = Space.load("cooking");
cooking.loadAgent("pantry-manager");
await inspect();

// Delegate pantry check to pantry-manager via fork
const pantryFork = fork<{ pantrySnapshot: object; useSoonItems: string[] }>({
  instruction: "[model:XS] Read the current pantry pin. Identify items expiring within 3 days (useSoonItems). Return { pantrySnapshot, useSoonItems }.",
  tokenBudget: 2000,
});
await inspect(pantryFork);

const pantryData = pantryFork as unknown as { pantrySnapshot: object; useSoonItems: string[] };
__flow.finish("check_pantry", pantryData);

if ((pantryData?.useSoonItems ?? []).length > 0) {
  display(
    <Markdown>{"⚠ **Use soon:** " + pantryData.useSoonItems.join(", ")}</Markdown>,
    { id: "use-soon" },
  );
}

pin("people");
pin("mealTypes");
pin("restrictions");
checkpoint("after-constraints");

await inspect(pantryData);
```
