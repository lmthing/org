# Phase 3 — Build shopping list and submit

```ts
// ── shopping_list ────────────────────────────────────────────────────────
__flow.start("shopping_list");

// Aggregate all main ingredients from approved meals
const allIngredients = (__approvedMeals as ApprovedMeal[])
  .flatMap(m => m.mainIngredients ?? []);

// Parse them into structured form
const parsed = await parseIngredients(allIngredients.join("\n")) as ParsedIngredient[];

// Diff against pantry — fork one worker per category for parallel processing
const categories = ["produce", "protein", "dairy", "pantry-staples", "other"];
const categoryForks = categories.map(cat =>
  fork<ShoppingItem[]>({
    instruction: `[model:XS] From this ingredient list: ${JSON.stringify(parsed)}, filter those in the "${cat}" category. Diff against pantry: ${JSON.stringify((__pantryData as any)?.pantrySnapshot ?? {})}. Return only items that need to be bought as ShoppingItem[]: { ingredient, quantity, unit, category }.`,
    tokenBudget: 2000,
  })
);

await inspect(...categoryForks);

const shoppingList = (categoryForks as unknown as ShoppingItem[][])
  .flat()
  .filter(Boolean);

display(
  <Table data={shoppingList.map(i => ({ Category: i.category, Item: i.ingredient, Amount: `${i.quantity ?? ""} ${i.unit ?? ""}`.trim() }))} />,
  { id: "shopping-list" },
);

__flow.finish("shopping_list", { shoppingList });

// ── submit ───────────────────────────────────────────────────────────────
__flow.start("submit");
submitMealPlan({ meals: __approvedMeals as object[], shoppingList });
__flow.finish("submit");
```

No `inspect()` in the final phase — call `submitMealPlan` directly. The session terminates as soon as the sink fires.
