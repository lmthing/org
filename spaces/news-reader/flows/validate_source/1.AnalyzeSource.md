# Phase 1 — Analyze source

Fetch domain info and article content for evaluation.

```ts
const __flow = tasklist("validate_source", { /* DAG from harness */ });

__flow.start("analyze");
const targetUrl = /* user-provided URL or domain */;
const domain = new URL(targetUrl.startsWith("http") ? targetUrl : `https://${targetUrl}`).hostname.replace(/^www\./, "");

const domainInfo = await getDomainInfo(domain);

let articleContent = "";
let claims: Array<{ claim: string; type: string }> = [];
try {
  const article = await fetchArticle(targetUrl, { byteBudget: 30000, extractMetadata: true });
  articleContent = article.markdown;
  const entities = await extractEntities(articleContent, {
    types: ["person", "organisation", "location", "date", "money"],
    maxEntities: 15,
  });
  // Convert entities and key statements into verifiable claims
} catch {
  articleContent = "(Could not fetch article content — domain-level assessment only)";
}

__flow.finish("analyze", { domainInfo, articleContent, claims });

await inspect(domainInfo, articleContent, claims);
```
