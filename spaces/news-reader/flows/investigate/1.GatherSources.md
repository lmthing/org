# Phase 1 — Gather sources

Search broadly, triage by credibility, then read the top articles.

// ── read ─────────────────────────────────────────────────────────────────
__flow.start("read");
const excerpts: Array<{ sourceId: string; title: string; url: string; text: string | null; error?: string; credibility: number }> = [];
for (let i = 0; i < topUrls.length; i++) {
  const url = topUrls[i]!;
  const sourceId = `source-${i + 1}`;
  try {
    const doc = await fetchArticle(url, { byteBudget: 30000 });
    excerpts.push({ sourceId, title: doc.title, url: doc.finalUrl, text: doc.markdown, credibility: 0 });
  } catch (e) {
    excerpts.push({ sourceId, title: url, url, text: null, error: String(e), credibility: 0 });
  }
}
__flow.finish("read", { excerpts });

await inspect(topUrls, excerpts);
