# Phase 1 — Manage feeds

Load subscriptions, execute the requested action, and persist.

```ts
const __flow = tasklist("feed_management", { /* DAG from harness */ });

// ── load ─────────────────────────────────────────────────────────────────
__flow.start("load");
const feeds: string[] = JSON.parse(
  (await Space.current().read("feeds.json")) ?? "[]"
);
__flow.finish("load", { feeds });

// ── execute ──────────────────────────────────────────────────────────────
__flow.start("execute");
let updatedFeeds = [...feeds];
let action = /* user-requested action: "add" | "remove" | "import" | "list" */;
let message = "";

if (action === "add") {
  const newUrl = /* URL provided by user */;
  // Validate by attempting a fetch
  try {
    await fetchRSS(newUrl, { maxItems: 1 });
    if (!updatedFeeds.includes(newUrl)) {
      updatedFeeds.push(newUrl);
      message = `Added ${newUrl} to subscriptions.`;
    } else {
      message = `${newUrl} is already subscribed.`;
    }
  } catch {
    message = `Could not validate ${newUrl} as a valid RSS feed. Please check the URL.`;
  }
} else if (action === "remove") {
  const targetUrl = /* URL provided by user */;
  updatedFeeds = updatedFeeds.filter(f => f !== targetUrl);
  message = `Removed ${targetUrl} from subscriptions.`;
} else if (action === "import") {
  const opmlSource = /* OPML file path or URL */;
  try {
    const opml = await readOpml(opmlSource);
    const newUrls = opml.outlines.map(o => o.xmlUrl).filter(u => !updatedFeeds.includes(u));
    updatedFeeds.push(...newUrls);
    message = `Imported ${newUrls.length} feeds from OPML (${opml.outlines.length} total found, ${newUrls.length} new).`;
  } catch (e) {
    message = `Failed to import OPML: ${String(e)}`;
  }
} else {
  message = `Current subscriptions (${updatedFeeds.length}):\n${updatedFeeds.map(f => `- ${f}`).join("\n")}`;
}
__flow.finish("execute", { feeds: updatedFeeds, action, message });

// ── submit ───────────────────────────────────────────────────────────────
__flow.start("submit");
// Persist the updated feed list
checkpoint("before-feed-update");
Space.current().write("feeds.json", JSON.stringify([...new Set(updatedFeeds)], null, 2));
submitFeedAction({ action, feeds: [...new Set(updatedFeeds)], message });
__flow.finish("submit");
```
