# Phase 1 — Collect and extract signals

Search across time windows and extract entities for trend detection.

```ts
const __flow = tasklist("trend_analysis", { /* DAG from harness */ });

// ── collect ──────────────────────────────────────────────────────────────
__flow.start("collect");
const topic = /* user-provided topic */;
const windows = [
  { label: "past_24h", freshness: "day" as const },
  { label: "past_week", freshness: "week" as const },
  { label: "past_month", freshness: "month" as const },
];

const timeWindows: Array<{ label: string; articleCount: number; articles: any[] }> = [];
for (const w of windows) {
  try {
    const results = await searchNews(topic, { freshness: w.freshness, topK: 15 });
    timeWindows.push({ label: w.label, articleCount: results.length, articles: results });
  } catch {
    timeWindows.push({ label: w.label, articleCount: 0, articles: [] });
  }
}
__flow.finish("collect", { timeWindows, topic });

// ── extract_signals ──────────────────────────────────────────────────────
__flow.start("extract_signals");
const allText = timeWindows.flatMap(w => w.articles.map(a => `${a.title} ${a.snippet}`)).join(" ");
const entities = await extractEntities(allText, {
  types: ["person", "organisation", "location", "technology", "event"],
  maxEntities: 30,
});
// Sort by frequency — entities with high count are trending
const entityFrequency = entities
  .sort((a, b) => b.count - a.count)
  .slice(0, 20);
__flow.finish("extract_signals", { entities, entityFrequency });

await inspect(timeWindows, entityFrequency);
```
