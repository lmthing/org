# Phase 2 — Categorise and fetch details

Group articles into topic categories, then fetch full content for featured items.

```ts
// ── categorise ───────────────────────────────────────────────────────────
__flow.start("categorise");
const articles = /* from triage result */;
// Group articles into categories based on their titles, descriptions, and any
// existing category metadata from the RSS items.
const categories: Array<{
  name: string;
  articles: any[];
  count: number;
}> = [
  // Build category buckets: politics, technology, business, science, health,
  // world, sports, culture, other. Assign each article to its best-fit category.
];
__flow.finish("categorise", { categories });

// ── fetch_details ────────────────────────────────────────────────────────
__flow.start("fetch_details");
// For each category, pick the top (most recent + highest credibility) article
// and fetch its full content.
const featured: Array<{
  category: string;
  title: string;
  url: string;
  content: string;
  source: string;
  pubDate: string;
}> = [];
for (const cat of categories) {
  if (cat.articles.length === 0) continue;
  const top = cat.articles[0];
  try {
    const article = await fetchArticle(top.link, { byteBudget: 20000 });
    featured.push({
      category: cat.name,
      title: article.title,
      url: article.finalUrl,
      content: article.markdown.slice(0, 2000),
      source: new URL(article.finalUrl).hostname.replace(/^www\./, ""),
      pubDate: top.pubDate,
    });
  } catch (e) {
    // Skip articles that fail to fetch — use the snippet instead
    featured.push({
      category: cat.name,
      title: top.title,
      url: top.link,
      content: top.description,
      source: new URL(top.link).hostname.replace(/^www\./, ""),
      pubDate: top.pubDate,
    });
  }
}
__flow.finish("fetch_details", { featured });

await inspect(categories, featured);
```

End the cycle here. Cycle 3 receives `categories` and `featured` for composition.
