# Phase 1 — Fetch feeds and triage

Load the subscription list, fetch all feeds, deduplicate and rank.

```ts
const __flow = tasklist("morning_briefing", { /* DAG from harness */ });

// ── fetch_feeds ──────────────────────────────────────────────────────────
__flow.start("fetch_feeds");
const feedUrls: string[] = JSON.parse(
  (await Space.current().read("feeds.json")) ?? "[]"
);
if (feedUrls.length === 0) {
  __flow.finish("fetch_feeds", { items: [], errors: ["No feeds subscribed. Use the feed_management flow to add feeds first."] });
} else {
  const since = new Date(Date.now() - 48 * 3600 * 1000).toISOString();
  const allItems: any[] = [];
  const errors: string[] = [];
  for (const url of feedUrls) {
    try {
      const feed = await fetchRSS(url, { since, maxItems: 20, includeContent: false });
      allItems.push(...feed.items.map(item => ({ ...item, feedUrl: url, feedTitle: feed.title })));
    } catch (e) {
      errors.push(`${url}: ${String(e)}`);
    }
  }
  __flow.finish("fetch_feeds", { items: allItems, errors });
}

__flow.start("triage");
const rawItems = /* from fetch_feeds result */;
// Deduplicate by URL (canonicalize). Sort by pubDate descending.
// For each unique domain, check credibility via getDomainInfo.
// Keep top 30 articles, weighted by recency + credibility.
const topArticles: any[] = /* deduplicated, ranked array */;
const stats = { total: rawItems.length, afterDedup: topArticles.length, feedsChecked: feedUrls.length };
__flow.finish("triage", { topArticles, stats });

await inspect(topArticles, stats);
```

End the cycle here. Cycle 2 receives `topArticles` and `stats`.
